cmake_minimum_required(VERSION 3.16)
project(ecm C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SOEM_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/SOEM)
add_library(ecm SHARED csrc/ecm.c)
set_target_properties(ecm PROPERTIES C_VISIBILITY_PRESET hidden PREFIX "lib")
target_link_libraries(ecm PRIVATE soem)
install(TARGETS ecm LIBRARY DESTINATION ethercat_master RUNTIME DESTINATION ethercat_master)
